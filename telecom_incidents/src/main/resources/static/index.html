Примеры запросов (нажмите, чтобы развернуть) 
# Создать инцидент
curl -X POST http://localhost:8081/api/incidents \
  -H "Content-Type: application/json" \
  -d '{"clientName":"Тестовый клиент","serviceType":"Интернет","problemType":"Клиентская","macroSegment":"B2B","address":"г. Москва"}'

# Получить список инцидентов
curl http://localhost:8081/api/incidents

# Получить задания
curl http://localhost:8081/api/tasks

# Получить правила для задания (подставьте реальный taskId)
curl http://localhost:8081/api/tasks/{taskId}/rules

# Завершить задание
curl -X POST http://localhost:8081/api/tasks/{taskId}/complete \
  -H "Content-Type: application/json" \
  -d '{"ruleId":14527,"comment":"Проблема решена"}'
